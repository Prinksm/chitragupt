CREATE TABLE IF NOT EXISTS  test_category (
    id SERIAL PRIMARY KEY,
    test_code INTEGER NOT NULL REFERENCES concept(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE IF NOT EXISTS  test_schedule (
    id SERIAL PRIMARY KEY,
    test_category_id INTEGER NOT NULL REFERENCES test_category(id) ON DELETE CASCADE,
    clinic_id INTEGER REFERENCES clinic(id) ON DELETE SET NULL,
    patient_id INTEGER NOT NULL REFERENCES patient(id) ON DELETE CASCADE,
    reminder_sent BOOLEAN DEFAULT FALSE,
    test_repeat VARCHAR(50), -- none, daily, weekly, monthly
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE IF NOT EXISTS  test_result (
    id SERIAL PRIMARY KEY,
    clinic_id INTEGER REFERENCES clinic(id) ON DELETE SET NULL,
    diagnostic_report_id INTEGER REFERENCES diagnostic_report(id) ON DELETE SET NULL,
    status VARCHAR(20) NOT NULL, -- registered, partial, preliminary, final, etc.
    issued_date TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE IF NOT EXISTS  test_reminder (
    id SERIAL PRIMARY KEY,
    test_schedule_id INTEGER NOT NULL REFERENCES test_schedule(id) ON DELETE CASCADE,
    sent_at TIMESTAMP,
    status VARCHAR(20) NOT NULL, -- sent, delivered, read, failed
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
