CREATE SEQUENCE IF NOT EXISTS test_category_id_seq START WITH 1;
CREATE TABLE IF NOT EXISTS  diagnosis.test_category (
    id BIGINT PRIMARY KEY DEFAULT nextval('test_category_id_seq'),
    test_code INTEGER NOT NULL REFERENCES codeable_concept.concept(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE SEQUENCE IF NOT EXISTS test_schedule_id_seq START WITH 1;
CREATE TABLE IF NOT EXISTS  diagnosis.test_schedule (
    id BIGINT PRIMARY KEY DEFAULT nextval('test_schedule_id_seq'),
    test_category_id INTEGER NOT NULL REFERENCES diagnosis.test_category(id) ON DELETE CASCADE,
    clinic_id INTEGER REFERENCES clinic(id) ON DELETE SET NULL,
    patient_id INTEGER NOT NULL REFERENCES patient.patient(id) ON DELETE CASCADE,
    reminder_sent BOOLEAN DEFAULT FALSE,
    test_repeat VARCHAR(50), -- none, daily, weekly, monthly
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE SEQUENCE IF NOT EXISTS test_result_id_seq START WITH 1;
CREATE TABLE IF NOT EXISTS  diagnosis.test_result (
    id BIGINT PRIMARY KEY DEFAULT nextval('test_result_id_seq'),
    clinic_id INTEGER REFERENCES patient.clinic(id) ON DELETE SET NULL,
    diagnostic_report_id INTEGER REFERENCES diagnosis.diagnostic_report(id) ON DELETE SET NULL,
    status VARCHAR(20) NOT NULL, -- registered, partial, preliminary, final, etc.
    issued_date TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE SEQUENCE IF NOT EXISTS test_reminder_id_seq START WITH 1;
CREATE TABLE IF NOT EXISTS  diagnosis.test_reminder (
    id BIGINT PRIMARY KEY DEFAULT nextval('test_reminder_id_seq'),
    test_schedule_id INTEGER NOT NULL REFERENCES diagnosis.test_schedule(id) ON DELETE CASCADE,
    sent_at TIMESTAMP,
    status VARCHAR(20) NOT NULL, -- sent, delivered, read, failed
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
